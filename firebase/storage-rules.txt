rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    function owns(uid) { return request.auth != null && request.auth.uid == uid; }
    match /users/{uid}/profile/{file} {
      allow read: if owns(uid);
      allow write: if owns(uid) && request.resource.size < 5 * 1024 * 1024 && request.resource.contentType.matches('image/(jpeg|png|webp)');
    }
    match /{allPaths=**} { allow read, write: if false; }
  }
}
