rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function owns(uid) { return signedIn() && request.auth.uid == uid; }
    function validProfile() {
      let d = request.resource.data;
      return d.keys().hasOnly(['displayName','username','country','language','timezone','dateOfBirth','createdAt','roles','goals'])
        && d.keys().hasAll(['displayName','username','country','language','timezone','dateOfBirth','createdAt','roles','goals'])
        && d.displayName is string && d.displayName.size() >= 1 && d.displayName.size() <= 50
        && d.username is string && d.username.matches('^[a-z0-9_]{3,24}$')
        && d.country is string && d.country.size() <= 80
        && d.language is string && d.language.size() <= 12
        && d.timezone is string && d.timezone.size() <= 80
        && d.dateOfBirth is string && d.dateOfBirth.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && d.roles is list && d.roles.size() <= 10
        && d.goals is list && d.goals.size() <= 20
        && d.createdAt == request.time;
    }
    function validProfileUpdate() {
      let d = request.resource.data;
      return d.diff(resource.data).affectedKeys().hasOnly(['displayName','country','language','timezone','roles','goals','updatedAt'])
        && d.displayName is string && d.displayName.size() >= 1 && d.displayName.size() <= 50
        && d.country is string && d.country.size() <= 80
        && d.language is string && d.language.size() <= 12
        && d.timezone is string && d.timezone.size() <= 80
        && d.roles is list && d.roles.size() <= 10
        && d.goals is list && d.goals.size() <= 20;
    }
    match /users/{uid} {
      // Profile records, including date of birth, are private to the account owner.
      allow read: if owns(uid);
      allow create: if owns(uid) && validProfile();
      allow update: if owns(uid) && validProfileUpdate();
      allow delete: if owns(uid);
      match /settings/{doc} {
        allow read: if owns(uid);
        allow create, update: if owns(uid)
          && request.resource.data.keys().hasOnly(['systems','theme','locale','units','privacy','notifications','updatedAt']);
        allow delete: if owns(uid);
      }
      match /avatars/{doc} {
        allow read: if owns(uid);
        allow create, update: if owns(uid)
          && request.resource.data.keys().hasOnly(['skinTone','hairStyle','hairColor','eyeColor','clothing','accessories','animationPack','updatedAt']);
        allow delete: if owns(uid);
      }
      match /{collection}/{doc} {
        // Other game records and reward balances stay denied until server-validated write paths exist.
        allow read, delete: if owns(uid) && collection in ['quests','chores','campaigns','notifications','relationships','userStats'];
        allow create, update: if false;
      }
    }
    match /publicCatalog/{document=**} { allow read: if signedIn(); allow write: if false; }
    match /{document=**} { allow read, write: if false; }
  }
}
