rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function owns(uid) { return signedIn() && request.auth.uid == uid; }
    function validText(value, max) { return value is string && value.size() > 0 && value.size() <= max; }
    // Private user-owned records. Keep privileged moderation and catalog writes server-side.
    match /users/{uid} {
      allow read: if owns(uid);
      allow create: if owns(uid) && request.resource.data.keys().hasOnly(['displayName','username','country','language','timezone','createdAt','roles','goals']);
      allow update: if owns(uid) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['displayName','country','language','timezone','roles','goals','updatedAt']);
      allow delete: if owns(uid);
      match /{collection}/{doc} {
        allow read, delete: if owns(uid);
        allow create, update: if owns(uid) && collection in ['quests','chores','campaigns','settings','avatars','notifications','userStats','relationships'];
      }
    }
    match /publicCatalog/{document=**} { allow read: if signedIn(); allow write: if false; }
    match /{document=**} { allow read, write: if false; }
  }
}
